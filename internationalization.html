    // =========================================================================
// 16-i18n/src/i18n.ts - the vue-i18n configuration
// -------------------------------------------------------------------------
// view-i18n is the translation library for Vue.js. It is provided with:
//    - one DICTIONNAIRE per language (files in `locales/fr.json`, `locales/en.json`),
//      where each text has a CLÉ ("ACCUEIL.BONJOUR");
//    - the source language (here: the one stored in localStorage; otherwise
//      the browser’s language, otherwise French);
//    - date and number formats specific to each language.
// In components, t('ACCUEIL.BONJOUR', { name: 'Marie' }) returns the
// text in the language COURANTE; changing the language updates the entire page.
//
// On the client side, this works on the same principle as the NestJS course (dictionaries
// i18n/*.json + {{t}} helper) — but the translation now takes place in
// the browser, and changing the language no longer requires any server interaction.
// =========================================================================

import { createI18n } from 'vue-i18n';
import fr from './locales/fr.json';
import en from './locales/en.json';

export type Langue = 'fr' | 'en';
export const CLE_LANGUE = 'exemple16.langue';

function langueDeDepart(): Langue {
  const memorisee = localStorage.getItem(CLE_LANGUE);
  if (memorisee === 'fr' || memorisee === 'en') {
    return memorisee;
  }
  return navigator.language.startsWith('en') ? 'en' : 'fr';
}

// message type: that of the French dictionary (the “template”)
type Messages = typeof fr;

export const i18n = createI18n<[Messages], Langue>({
  legacy: false, // the "Composition" (API), that of <script setup>
  locale: langueDeDepart(),
  fallbackLocale: 'fr', // a key missing in English: we use the French one
  messages: { fr, en },
   // d(date, 'long') and n(number, 'euros'): language-specific formats
  datetimeFormats: {
    fr: { long: { dateStyle: 'full' } },
    en: { long: { dateStyle: 'full' } },
  },
  numberFormats: {
    fr: { euros: { style: 'currency', currency: 'EUR' } },
    en: { euros: { style: 'currency', currency: 'EUR' } },
  },
});
    • line 19: [Vite] can import a JSON file: this returns an object;
    • lines 25–31: the language stored in [localStorage], otherwise the browser’s language ([navigator.language]), otherwise French;
    • lines 36–50: the configuration object:
      • [legacy: false]: the API “Composition” ([useI18n]), that of [<script setup>];
      • [fallbackLocale: 'fr']: a key missing from the English dictionary is taken from the French dictionary;
      • [datetimeFormats], [numberFormats]: date and number formats specific to each language.

    <!--
  =========================================================================
  16-i18n/src/App.vue - une page traduite
  -------------------------------------------------------------------------
    - t('CLE')                  un texte ;
    - t('CLE', { nom: ... })    un texte à paramètres ({nom} dans le dictionnaire) ;
    - t('CLE', n)               un pluriel : "aucun | un | {n}" selon n ;
    - d(date, 'long')           une date, au format de la langue ;
    - n(nombre, 'euros')        un nombre, au format de la langue ;
    - locale                    la langue courante : la modifier retraduit tout.
  =========================================================================
-->
<script setup lang="ts">
import { ref, watch } from 'vue';
import { useI18n } from 'vue-i18n';
import { CLE_LANGUE, type Langue } from './i18n';

const { t, d, n, locale } = useI18n();
const nombre = ref(2);

// Every time the language changes: it is saved, and the page is updated
// the page’s lang attribute (useful for screen readers and proofreaders)
watch(
  locale,
  (langue) => {
    localStorage.setItem(CLE_LANGUE, langue);
    document.documentElement.lang = langue;
    document.title = t('APP.TITRE');
  },
  { immediate: true },
);

function choisir(langue: Langue): void {
  locale.value = langue;
}

// what the server would return: RdvMedecins (see the chapter on the black box)
const reponsesServeur = [
  { statusCode: 409, cle: 'ERRORS.LOGIN_TAKEN' },
  { statusCode: 429, cle: 'ERRORS.TOO_MANY_ATTEMPTS', params: { minutes: '15' } },
];
</script>

<template>
  <div class="container py-4" style="max-width: 720px">
    <div class="d-flex justify-content-between align-items-center mb-3">
      <h1 class="h3 mb-0">{{ t('APP.TITRE') }}</h1>
      <div class="btn-group btn-group-sm" role="group" :aria-label="t('APP.LANGUE')">
        <button class="btn" :class="locale === 'fr' ? 'btn-primary' : 'btn-outline-primary'" @click="choisir('fr')">FR</button>
        <button class="btn" :class="locale === 'en' ? 'btn-primary' : 'btn-outline-primary'" @click="choisir('en')">EN</button>
      </div>
    </div>

    <div class="card p-3 mb-3">
      <p>{{ t('ACCUEIL.BONJOUR', { nom: 'Marie' }) }}</p>
      <p>{{ t('ACCUEIL.AUJOURDHUI', { date: d(new Date(), 'long') }) }}</p>
      <p>{{ t('ACCUEIL.PRIX', { prix: n(1234.5, 'euros') }) }}</p>
       <!-- t(key, number) selects the correct plural form -->
      <p class="fw-bold">{{ t('ACCUEIL.RENDEZ_VOUS', nombre) }}</p>
      <div class="d-flex gap-2">
        <button class="btn btn-sm btn-outline-success" @click="nombre++">{{ t('ACCUEIL.AJOUTER') }}</button>
        <button class="btn btn-sm btn-outline-secondary" :disabled="nombre === 0" @click="nombre--">{{ t('ACCUEIL.RETIRER') }}</button>
      </div>
    </div>

    <div class="card p-3">
      <h2 class="h6">{{ t('SERVEUR.TITRE') }}</h2>
      <p>{{ t('SERVEUR.EXPLICATION') }}</p>
      <ul class="mb-0">
        <li v-for="r in reponsesServeur" :key="r.cle">
          <code>{{ JSON.stringify(r) }}</code><br>
           <!-- the server key EST is a dictionary key; params provides its parameters -->
          &rarr; {{ t(r.cle, r.params ?? {}) }}
        </li>
      </ul>
    </div>
  </div>
</template>
    • line 18: translation tools:
      • [t(clé, paramètres)]: a text;
      • [t(clé, nombre)]: a plural;
      • [d(date, 'long')]: a date, in the language's format;
      • [n(nombre, 'euros')]: a number, in the language's format;
      • [locale]: the current language, a [ref]. Changing it re-translates the entire page;
    • lines 23–30: Each time the language is changed, the selection is saved and the page’s [lang] attribute is updated (useful for screen readers and spell-checkers);
    • lines 38–41: two error responses, as returned by the server [RdvMedecins];
    • line 73: the key sent by the server is a dictionary key; its parameters are those of the message.
