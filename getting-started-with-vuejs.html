    // =========================================================================
// vite.config.ts - the Vite configuration, common to the TOUS examples.
// -------------------------------------------------------------------------
// Vite is both:
//    - a development server (npm run dev): it compiles on the fly the
//      .vue and .ts files requested by the browser, and reloads the page
//      (or just the modified component) with every save;
//    - a build tool (npm run build): it generates, in the dist/ directory,
//      optimized .js and .css files to be deployed to a web server.
//
// Each example is a small, standalone application with its own
// index.html: 01-first-application/index.html, 02-responsiveness/index.html...
// A single server quickly serves them all: http://localhost:5173/ displays the
// the table of contents (index.html for this folder), http://localhost:5173/01-premiere-application/
// the first example, etc.
// =========================================================================

import { existsSync, readdirSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { resolve } from 'node:path';
import { defineConfig } from 'vite';
import vue from '@vitejs/plugin-vue';

const racine = fileURLToPath(new URL('.', import.meta.url));

// the example folders: those whose names begin with two digits
const exemples = readdirSync(racine).filter((d) => /^\d\d-/.test(d) && existsSync(resolve(racine, d, 'index.html')));

export default defineConfig({
   // the plugin that teaches Vite to compile .vue files
  plugins: [vue()],

  server: {
    port: 5173,
     // PROXY: any browser request to /api/... is forwarded by
     // Vite to the JSON RdvMedecins server (port 8080), and the response is then sent back
     // to the browser. From the browser’s perspective, the page and API therefore originate from
     // (http://localhost:5173): no CORS to configure, and the cookie for
     // token JWT circulates normally (see the chapter on the black box).
    proxy: {
       '/api': 'http://localhost:8080',
    },
  },

   // `npm run build`: an entry page, for example, plus the table of contents
  build: {
    rollupOptions: {
      input: Object.fromEntries([
        ['sommaire', resolve(racine, 'index.html')],
        ...exemples.map((d) => [d, resolve(racine, d, 'index.html')]),
      ]),
    },
  },
});
    • line 31: the extension that instructs [Vite] to compile the [.vue] files. A browser cannot read a [.vue] file: when it requests one, [Vite] converts it on the fly into a standard [JavaScript] file;
    • line 34: the development server’s port;
    • lines 40–41: the proxy, which will only be used starting with Example 17: any request from the browser to an address beginning with [/api] is forwarded to the server JSON [RdvMedecins], on port 8080. We’ll come back to this;
    • Lines 27 and 46–53: [npm run build] must know all the entry pages for the application. It is provided with the index and the [index.html] file for each folder whose name begins with two digits.

    <!--
  =========================================================================
  01-premiere-application/src/App.vue - un premier composant
  -------------------------------------------------------------------------
  Un fichier .vue (Single File Component) regroupe les trois facettes d'un
  composant :
    - <script setup> : les données et le code du composant (TypeScript) ;
    - <template>     : le HTML à afficher, avec des {{ ... }} ;
    - <style>        : son CSS (ici "scoped" : il ne s'applique qu'à lui).
  =========================================================================

<script setup lang="ts">
import { ref } from 'vue';

// "ordinary" data: the template can display it
const titre = 'Bienvenue dans Vue.js';
const langages = ['TypeScript', 'HTML', 'CSS'];

// ref(...): a RÉACTIVE data item. When its value (.value) changes, Vue updates
// updates the parts of the page that display it
const compteur = ref(0);
const heure = ref(new Date().toLocaleTimeString());

// a function called by the button in the template
function incrementer(): void {
  compteur.value++;
  heure.value = new Date().toLocaleTimeString();
}
</script>

<template>
  <div class="container py-4">
     <!-- {{ expression }}: the value of the expression, inserted into the page -->
    <h1 class="h3">{{ titre }}</h1>
    <p>Cette page est fabriquée dans le navigateur, par Vue.js, à partir de trois fichiers : index.html, main.ts et App.vue.</p>
    <p>On y écrit du {{ langages.join(', ') }} ({{ langages.length }} langages).</p>

     <!-- @click: the function to call when clicked (see example 04) -->
    <button class="btn btn-primary" @click="incrementer">Cliquez-moi</button>
     <!-- in the template, no .value: Vue "unpacks" the refs on its own -->
    <p class="mt-3 compteur">Vous avez cliqué {{ compteur }} fois (dernier clic : {{ heure }}).</p>
  </div>
</template>

<style scoped>
/* "scoped": this rule applies only to elements of the CE component */
.compteur {
  font-weight: bold;
  color: #0d6efd;
}
</style>

    A [.vue] file (Single File Component, or SFC) combines the three facets of a component:

    • lines 12–29: the component code, in [TypeScript] ([lang="ts"]). With [setup], all variables and functions declared at the top level of the script are directly usable in the template;
      • lines 16–17: two ordinary data items;
      • line 21: [ref(0)] creates a reactive data item with an initial value of 0. Its value is read and modified via its property [value]. When it changes, [Vue.js] updates all locations on the page where it is displayed;
      • lines 25–27: a function that modifies two reactive data fields. It does not alter the page: [Vue.js] will update it;
    • lines 31–43: the component’s HTML:
      • line 34: the [{{ expression }}] syntax (the “mustaches”) inserts the value of a [TypeScript] expression into the page. It is escaped: if [titre] contained [<b>], [<b>] would be displayed without the tag being interpreted;
      • line 36: the expression can be anything: a method call, a calculation;
      • line 39: clicking the button calls the function [incrementer] (see example 04);
      • line 41: in the template, you write [compteur], not [compteur.value]: [Vue.js] automatically “unpacks” the [ref];
    • lines 45–51: the component’s CSS. With [scoped], its rules apply only to elements of the CE component: [Vite] adds a unique attribute to its elements ([data-v-xxxx]) and includes it in the selectors.

    <!--
  =========================================================================
  02-reactivite/src/App.vue - ref, reactive, computed, watch
  -------------------------------------------------------------------------
  Un petit calculateur de prix TTC :
    - ref      : une valeur réactive simple (un nombre, une chaîne...) ;
    - reactive : un OBJET réactif (toutes ses propriétés le sont) ;
    - computed : une valeur CALCULÉE à partir d'autres, recalculée
                 automatiquement quand elles changent ;
    - watch    : du code exécuté quand une valeur change (un "effet").
  =========================================================================
-->
<script setup lang="ts">
import { computed, reactive, ref, watch } from 'vue';

// ref: its value is read and written using .value (in the TypeScript code)
const quantite = ref(1);

// reactive: an object whose properties are read and written directly
const article = reactive({
  libelle: 'Stylo',
  prixHT: 2.5,
  tauxTVA: 20, // in %
});

// computed: never assigned directly, always recalculated. The view knows
// which data it depends on (quantity, article.prixHT, article.tauxTVA):
// it recalculates it when any of these values change, and only in that case
const totalHT = computed(() => quantite.value * article.prixHT);
const montantTVA = computed(() => (totalHT.value * article.tauxTVA) / 100);
const totalTTC = computed(() => totalHT.value + montantTVA.value);

// a euro amount formatter, in the French style (1,234.50 €)
const euros = new Intl.NumberFormat('fr-FR', { style: 'currency', currency: 'EUR' });

// watch: every time the total TTC changes, a record is kept
const historique = ref<string[]>([]);
watch(totalTTC, (nouveau, ancien) => {
  historique.value.unshift(`${euros.format(ancien)} -> ${euros.format(nouveau)}`);
  historique.value = historique.value.slice(0, 5); // the last 5
});

function reinitialiser(): void {
  quantite.value = 1;
  article.prixHT = 2.5;
  article.tauxTVA = 20;
}
</script>

<template>
  <div class="container py-4">
    <h1 class="h3 mb-3">Exemple 02 - la réactivité</h1>
    <div class="row g-4">
      <div class="col-md-6">
        <div class="card p-3">
           <!-- v-model links a field to data, in both directions (see example 05) -->
          <label class="form-label">Article</label>
          <input v-model="article.libelle" class="form-control mb-2">
          <label class="form-label">Prix unitaire HT</label>
          <input v-model.number="article.prixHT" type="number" step="0.1" min="0" class="form-control mb-2">
          <label class="form-label">Quantité</label>
          <input v-model.number="quantite" type="number" min="0" class="form-control mb-2">
          <label class="form-label">Taux de TVA</label>
          <select v-model.number="article.tauxTVA" class="form-select mb-3">
            <option :value="5.5">5,5 %</option>
            <option :value="10">10 %</option>
            <option :value="20">20 %</option>
          </select>
          <button class="btn btn-outline-secondary" @click="reinitialiser">Réinitialiser</button>
        </div>
      </div>
      <div class="col-md-6">
        <div class="card p-3">
          <h2 class="h5">{{ quantite }} x {{ article.libelle }}</h2>
          <table class="table mb-0">
            <tbody>
              <tr><td>Total HT</td><td class="text-end">{{ euros.format(totalHT) }}</td></tr>
              <tr><td>TVA ({{ article.tauxTVA }} %)</td><td class="text-end">{{ euros.format(montantTVA) }}</td></tr>
              <tr class="fw-bold"><td>Total TTC</td><td class="text-end">{{ euros.format(totalTTC) }}</td></tr>
            </tbody>
          </table>
        </div>
        <div class="card p-3 mt-3">
          <h2 class="h6">Historique du total TTC (watch)</h2>
          <ul class="mb-0">
            <li v-for="(ligne, i) in historique" :key="i">{{ ligne }}</li>
          </ul>
        </div>
      </div>
    </div>
  </div>
</template>
    • line 17: [ref] accepts a simple value (number, string, boolean);
    • lines 20–24: [reactive] makes an entire object reactive: each of its properties is reactive. They are read and written directly ([article.prixHT]), without [.value];
    • lines 29–31: [computed] defines a value calculated from other values. [Vue.js] stores the data read during the calculation ([quantite], [article.prixHT]...) and only recalculates when one of them changes. It is never assigned a value: it is read ([totalHT.value]);
    • line 34: [Intl.NumberFormat], the number formatting tool integrated into [JavaScript]: [2.5] becomes “2.50 €”;
    • lines 38–41: [watch] executes code whenever data changes, receiving both the new and old values. It is used for actions such as saving data, sending a request, or writing to the browser’s storage... To display a derived value, we prefer [computed];
    • lines 43–46: Simply modify the data; all fields, totals, and the history will update accordingly;
    • lines 58–64: [v-model] links each field to a data point, in both directions (see example 05);
    • line 77: In the template, a [computed] is used in the same way as a [ref], without [.value];
    • line 86: a repeated list for each element of the [historique] array (see example 03).

  03-directives/src/App.vue - les directives de base
  -------------------------------------------------------------------------
  Une DIRECTIVE est un attribut spécial du template, préfixé par v- :
    - v-bind:attr="expr" (abrégé :attr="expr") : la valeur d'un attribut

<script setup lang="ts">
import { computed, ref } from 'vue';
// Import an image: Vite provides its URL (and, in production, an optimized copy
// optimized copy of the file, with a unique name)
import urlLogo from './assets/medecin.svg';

// An interface TypeScript describes the format of the objects in the list
interface Medecin {
  id: number;
  titre: string;
  prenom: string;
  nom: string;
  specialite: string;
  disponible: boolean;
}

const medecins = ref<Medecin[]>([
  { id: 1, titre: 'Mme', prenom: 'Marie', nom: 'PELISSIER', specialite: 'généraliste', disponible: true },
  { id: 2, titre: 'Mr', prenom: 'Paul', nom: 'TATOU', specialite: 'pédiatre', disponible: false },
  { id: 3, titre: 'Melle', prenom: 'Sophie', nom: 'DUPONT', specialite: 'généraliste', disponible: true },
]);

const filtre = ref(''); // search text
const seulementDisponibles = ref(false);
const afficherAide = ref(true);

// the displayed list: a computed field, recalculated when the filter changes
const medecinsAffiches = computed(() =>
  medecins.value.filter(
    (m) =>
      (!seulementDisponibles.value || m.disponible) &&
      `${m.prenom} ${m.nom} ${m.specialite}`.toLowerCase().includes(filtre.value.trim().toLowerCase()),
  ),
);
</script>

<template>
  <div class="container py-4">
    <h1 class="h3 mb-3">
       <!-- :src: the value of the src attribute is that of the variable urlLogo -->
      <img :src="urlLogo" alt="logo" width="32" height="32">
      Exemple 03 - les directives
    </h1>

    <div class="d-flex flex-wrap gap-3 align-items-center mb-3">
      <input v-model="filtre" class="form-control w-auto" placeholder="rechercher...">
      <label class="form-check-label">
        <input v-model="seulementDisponibles" type="checkbox" class="form-check-input"> disponibles seulement
      </label>
      <button class="btn btn-sm btn-outline-secondary" @click="afficherAide = !afficherAide">
        {{ afficherAide ? "Cacher l'aide" : "Montrer l'aide" }}
      </button>
    </div>

     <!-- v-show: the paragraph remains on the page, but is simply hidden -->

    <!--
  =========================================================================
  06-validation/src/App.vue - valider un formulaire dans le navigateur
  -------------------------------------------------------------------------
  Les erreurs sont des COMPUTED : elles sont recalculées à chaque frappe.
  On ne les affiche pas d'emblée (un formulaire vide couvert de rouge est
  désagréable) : seulement pour un champ déjà quitté ("touché"), ou pour
  tous les champs après une tentative d'envoi.

  Mêmes règles que le serveur RdvMedecins (formulaire "Créer un compte") :
  titre dans la liste, prénom et nom obligatoires (20 caractères au plus),
  login en minuscules, mot de passe de 6 caractères au moins, confirmation.

  ATTENTION : cette validation n'est qu'un CONFORT pour l'utilisateur (une
  réponse immédiate, sans aller-retour avec le serveur). Elle ne protège
  rien : n'importe qui peut envoyer au serveur une requête fabriquée à la
  main (curl...). Le serveur revalide donc TOUJOURS tout (cf. exemple 20).
  =========================================================================

<script setup lang="ts">
import { computed, reactive, ref } from 'vue';

const TITRES = ['Mr', 'Mme', 'Melle'];
const FORMAT_LOGIN = /^[a-z0-9._-]{1,30}$/;
const MOT_DE_PASSE_MIN = 6;

type Champ = 'titre' | 'prenom' | 'nom' | 'login' | 'password' | 'confirmation';

const saisie = reactive<Record<Champ, string>>({
  titre: 'Mr',
  prenom: '',
  nom: '',
  login: '',
  password: '',
  confirmation: '',
});

// fields the user has already exited (blur event)
const touches = reactive(new Set<Champ>());
const envoiTente = ref(false);
const succes = ref(false);

// the error for each field ('' = no error), recalculated with each keystroke
const erreurs = computed<Record<Champ, string>>(() => ({
  titre: TITRES.includes(saisie.titre) ? '' : 'Choisissez un titre dans la liste.',
  prenom: obligatoire(saisie.prenom, 'Le prénom') || maximum(saisie.prenom, 'Le prénom', 20),
  nom: obligatoire(saisie.nom, 'Le nom') || maximum(saisie.nom, 'Le nom', 20),
  login:
    obligatoire(saisie.login, 'Le login') ||
    (FORMAT_LOGIN.test(saisie.login.trim())
      ? ''
      : 'Lettres minuscules non accentuées, chiffres, point, tiret, souligné.'),
  password:
    obligatoire(saisie.password, 'Le mot de passe') ||
    (saisie.password.length < MOT_DE_PASSE_MIN ? `Au moins ${MOT_DE_PASSE_MIN} caractères.` : ''),
  confirmation: saisie.confirmation === saisie.password ? '' : 'La confirmation ne correspond pas au mot de passe.',
}));

const formulaireValide = computed(() => Object.values(erreurs.value).every((e) => e === ''));

function obligatoire(valeur: string, nom: string): string {
  return valeur.trim() === '' ? `${nom} est obligatoire.` : '';
}
function maximum(valeur: string, nom: string, max: number): string {
  return valeur.trim().length > max ? `${nom} ne doit pas dépasser ${max} caractères.` : '';
}

// the error at AFFICHER for a field: only if it has been touched, or afterward
// an attempt to submit
function erreurVisible(champ: Champ): string {
  return touches.has(champ) || envoiTente.value ? erreurs.value[champ] : '';
}

function envoyer(): void {
  envoiTente.value = true;
  succes.value = formulaireValide.value;
}
</script>
